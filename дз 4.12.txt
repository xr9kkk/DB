10. Выбрать названия факультетов, на которых количество студентов 2-го курса совпадает с количеством студентов на 2-м курсе факультета ПММ.
SELECT f.Название
FROM Факультеты f
JOIN Студенты s ON f.Id_факультета = s.Id_факультета
WHERE s.Курс = 2
GROUP BY f.Id_факультета, f.Название
HAVING COUNT(*) = (
    SELECT COUNT(*)
    FROM Студенты s2
    JOIN Факультеты f2 ON s2.Id_факультета = f2.Id_факультета
    WHERE s2.Курс = 2 AND f2.Название = 'ПММ'
)
AND f.Название != 'ПММ';

13. Выбрать все данные о студентах, учащихся на курсах, на которых менее 1000 студентов.
SELECT s.*
FROM Студенты s
WHERE s.Курс IN (
    SELECT Курс
    FROM Студенты
    GROUP BY Курс
    HAVING COUNT(*) < 1000
);
17. Выбрать курсы и группы без повторений тех факультетов, на которых преподается дисциплина «Базы данных».
SELECT DISTINCT s.Курс, s.Группа
FROM Студенты s
WHERE s.Id_факультета IN (
    SELECT DISTINCT f.Id_факультета
    FROM Факультеты f
    JOIN Преподаватели t ON f.Id_факультета = t.Id_факультета
    JOIN Успеваемость u ON t.Id_преподавателя = u.Id_преподавателя
    JOIN Дисциплины d ON u.Id_дисциплины = d.Id_дисциплины
    WHERE d.Название = 'Базы данных'
)
ORDER BY s.Курс, s.Группа;
26. Вывести в одном столбце фамилии, имена, отчества студентов и преподавателей. Для преподавателей указать количество
предметов, которые ведет преподаватель, для студентов – количество предметов, по которым студент имеет оценку «5» или «4».
Результат отсортировать по фамилии, имени, отчеству в лексикографическом порядке.
SELECT 
    CONCAT(Фамилия, ' ', Имя, ' ', Отчество) AS ФИО,
    Количество_предметов
FROM (
    SELECT 
        t.Id_преподавателя,
        t.Фамилия,
        t.Имя,
        t.Отчество,
        COUNT(DISTINCT u.Id_дисциплины) AS Количество_предметов
    FROM Преподаватели t
    LEFT JOIN Успеваемость u ON t.Id_преподавателя = u.Id_преподавателя
    GROUP BY t.Id_преподавателя, t.Фамилия, t.Имя, t.Отчество
    
    UNION ALL
    
    SELECT 
        s.Номер_Студ_Билета,
        s.Фамилия,
        s.Имя,
        s.Отчество,
        COUNT(DISTINCT u2.Id_дисциплины) AS Количество_предметов
    FROM Студенты s
    LEFT JOIN Успеваемость u2 ON s.Номер_Студ_Билета = u2.Номер_Студ_Билета
    WHERE u2.Оценка IN (5, 4) OR u2.Оценка IS NULL
    GROUP BY s.Номер_Студ_Билета, s.Фамилия, s.Имя, s.Отчество
) AS CombinedData
ORDER BY ФИО;
27. Выбрать два зимних месяца, в которые родилось больше
всего преподавателей и студентов. 
SELECT Месяц
FROM (
    SELECT 
        EXTRACT(MONTH FROM Дата_рождения) AS Месяц,
        COUNT(*) AS Количество,
        ROW_NUMBER() OVER (ORDER BY COUNT(*) DESC) AS rn
    FROM (
        SELECT Дата_рождения FROM Студенты
        WHERE EXTRACT(MONTH FROM Дата_рождения) IN (12, 1, 2)
        UNION ALL
        SELECT Дата_рождения FROM Преподаватели
        WHERE EXTRACT(MONTH FROM Дата_рождения) IN (12, 1, 2)
    ) AS Все_Дни_Рождения
    GROUP BY EXTRACT(MONTH FROM Дата_рождения)
) AS RankedMonths
WHERE rn <= 2
ORDER BY Месяц;
28. Выбрать названия дисциплин, которые на «5» сдали 2/3 от всех сдававших.
SELECT d.Название
FROM Дисциплины d
WHERE d.Id_дисциплины IN (
    SELECT u.Id_дисциплины
    FROM Успеваемость u
    GROUP BY u.Id_дисциплины
    HAVING 
        SUM(CASE WHEN u.Оценка = 5 THEN 1 ELSE 0 END) * 1.0 / COUNT(*) >= 2.0/3.0
);
